  <script>
    document.addEventListener("DOMContentLoaded", () => {
      // 1. Set Current Year in Footer
      document.getElementById("year").textContent = new Date().getFullYear();

      // 2. Hamburger Mobile Menu Toggle
      const hamburger = document.getElementById("hamburger");
      const navLinks = document.getElementById("navLinks");
      const navItems = navLinks.querySelectorAll("a");

      hamburger.addEventListener("click", () => {
        navLinks.classList.toggle("active");
        const icon = hamburger.querySelector("i");
        icon.classList.toggle("fa-bars");
        icon.classList.toggle("fa-xmark");
      });

      // Close mobile menu when clicking a link
      navItems.forEach((item) => {
        item.addEventListener("click", () => {
          navLinks.classList.remove("active");
          const icon = hamburger.querySelector("i");
          icon.classList.add("fa-bars");
          icon.classList.remove("fa-xmark");
        });
      });

      // 3. Automatic YouTube RSS Video Feed via rss2json API
      // Replace 'not avevale' with a valid 24-character YouTube Channel ID (e.g., 'UC_x5XG1OV2P6uZZ5FSM9Ttw')
      const CHANNEL_ID = "not avevale";
      const videoGrid = document.getElementById("videoGrid");

      async function fetchLatestVideos() {
        // Validate Channel ID before calling the API
        if (!CHANNEL_ID || CHANNEL_ID.trim().toLowerCase() === "not avevale" || !CHANNEL_ID.startsWith("UC")) {
          videoGrid.innerHTML = `
            <div class="feed-status">
              <i class="fa-brands fa-youtube"></i>
              <h3>YouTube Channel ID Not Configured</h3>
              <p>
                Channel ID is currently set to <strong>"${CHANNEL_ID}"</strong>. 
                Update the <code>CHANNEL_ID</code> variable in the JavaScript code with your 24-character YouTube Channel ID (starting with <code>UC...</code>) to automatically load your latest 3 videos.
              </p>
            </div>
          `;
          return;
        }

        const rssUrl = `https://www.youtube.com/feeds/videos.xml?channel_id=${encodeURIComponent(CHANNEL_ID)}`;
        const apiUrl = `https://api.rss2json.com/v1/api.json?rss_url=${encodeURIComponent(rssUrl)}`;

        try {
          const response = await fetch(apiUrl);
          const data = await response.json();

          if (data.status !== "ok" || !data.items || data.items.length === 0) {
            throw new Error("No videos found in RSS feed.");
          }

          // Take the latest 3 videos
          const latestVideos = data.items.slice(0, 3);
          videoGrid.innerHTML = "";

          latestVideos.forEach((item) => {
            // Parse Video ID dynamically from guid (format: "yt:video:VIDEO_ID")
            const videoId = item.guid && item.guid.includes(":")
              ? item.guid.split(":").pop()
              : "";

            // Construct dynamic YouTube URL and high-res thumbnail from parsed videoId
            const dynamicVideoUrl = `https://www.youtube.com/watch?v=${videoId}`;
            const dynamicThumbnail = `https://i.ytimg.com/vi/${videoId}/hqdefault.jpg`;

            // Format publish date
            const pubDate = new Date(item.pubDate).toLocaleDateString("en-US", {
              year: "numeric",
              month: "short",
              day: "numeric",
            });

            const card = document.createElement("a");
            card.href = dynamicVideoUrl;
            card.target = "_blank";
            card.rel = "noopener noreferrer";
            card.className = "video-card";

            card.innerHTML = `
              <div class="thumbnail-wrapper">
                <img src="${dynamicThumbnail}" alt="${item.title}" loading="lazy" />
                <div class="play-overlay">
                  <div class="play-icon">
                    <i class="fa-solid fa-play"></i>
                  </div>
                </div>
              </div>
              <div class="video-info">
                <h3 class="video-title">${item.title}</h3>
                <div class="video-meta">
                  <span><i class="fa-regular fa-calendar"></i> ${pubDate}</span>
                  <span style="color: var(--accent-color); font-weight: 600;">Watch Now <i class="fa-solid fa-arrow-up-right-from-square"></i></span>
                </div>
              </div>
            `;

            videoGrid.appendChild(card);
          });
        } catch (error) {
          console.error("Error fetching YouTube RSS feed:", error);
          videoGrid.innerHTML = `
            <div class="feed-status">
              <i class="fa-solid fa-triangle-exclamation"></i>
              <h3>Unable to Load Videos</h3>
              <p>Could not fetch videos from the YouTube RSS feed. Please verify the Channel ID and try again later.</p>
            </div>
          `;
        }
      }

      fetchLatestVideos();

      // 4. Contact Form Submission Handler
      const contactForm = document.getElementById("contactForm");
      const formFeedback = document.getElementById("formFeedback");

      contactForm.addEventListener("submit", (e) => {
        e.preventDefault();
        const name = document.getElementById("name").value.trim();

        formFeedback.style.display = "block";
        formFeedback.style.color = "#4ade80";
        formFeedback.innerHTML = `<i class="fa-solid fa-circle-check"></i> Thanks, <strong>${name}</strong>! Your message has been sent to FORHAD AI.`;

        contactForm.reset();

        setTimeout(() => {
          formFeedback.style.display = "none";
        }, 5000);
      });
    });
  </script>
